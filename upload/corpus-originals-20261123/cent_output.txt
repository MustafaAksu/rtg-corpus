import sympy as sp
z,Z=sp.symbols('z Z')  # Z = conj z
s3=sp.sqrt(3); mu=-sp.Rational(1,2)-2*sp.I*s3/3; mub=sp.conjugate(mu)
X=mu*z**2*Z; Xb=mub*Z**2*z
def bracket(Y,Yb):  # [X,Y] = DY.X - DX.Y  (complex field form)
    DYX=sp.diff(Y,z)*X+sp.diff(Y,Z)*Xb
    DXY=sp.diff(X,z)*Y+sp.diff(X,Z)*Yb
    return sp.expand(DYX-DXY)
for k in range(2,14):
    mons=[(a,k-a) for a in range(k+1)]
    cols=[]
    for (a,b) in mons:
        for unit in (1,sp.I):   # real basis: coefficient unit
            Y=unit*z**a*Z**b; Yb=sp.conjugate(unit)*Z**a*z**b
            B=sp.Poly(bracket(Y,Yb),z,Z)
            vec=[]
            for (a2,b2) in [(a2,k+2-a2) for a2 in range(k+3)]:
                cf=B.coeff_monomial(z**a2*Z**b2); cf=sp.nsimplify(cf)
                vec+=[sp.re(cf),sp.im(cf)]
            cols.append(vec)
    M=sp.Matrix(cols).T
    kd=M.shape[1]-M.rank()
    ker=M.nullspace()
    desc=[]
    for v in ker:
        terms=[]
        for i,(a,b) in enumerate(mons):
            c=v[2*i]+sp.I*v[2*i+1]
            if c!=0: terms.append(f"({sp.simplify(c)}) z^{a} zb^{b}")
        desc.append(" + ".join(terms))
    print(f"k={k}: dim ker ad_X3 = {kd}", desc)
