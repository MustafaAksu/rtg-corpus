# Exact (rational-arithmetic) six-step resonant cubic of the kappa center-manifold reduced map.
# Structure: f3(k) = A + N(k)/D(k), D=(k-2)(k^2-k+1), deg N<=3  (G3 affine in psi2; G2 kappa-free)
# => q(k)*D(k) is a polynomial of degree <=3 : 4 exact points determine it, extra points verify.
import sympy as sp
from sympy import Rational as R
a,b,c,d=sp.symbols('a b c d')
V=(a,b,c,d)
def tr(e,n=3):
    p=sp.Poly(sp.expand(e),*V); return sum(cf*sp.prod([v**m for v,m in zip(V,mo)]) for mo,cf in zip(p.monoms(),p.coeffs()) if sum(mo)<=n)
def T3(r,e):  # cubic Taylor of T
    return (r-e+e**3/6, r+r**2/2+r**3/6+e**3/6)
def psi2(k):
    D=(k-2)*(k**2-k+1)
    p0=-(k-1)**2*(a**2*k-2*a**2+2*a*b+a*c*k-2*a*c+a*d-b**2+b*c-b*d+c**2*k-2*c**2+2*c*d-d**2)/D
    p1=-(k-1)*(a**2*k**2-2*a**2*k+a**2+a*c*k**2-2*a*c*k+a*c-b**2-b*d+c**2*k**2-2*c**2*k+c**2-d**2)/D
    return (sp.expand(p0),sp.expand(p1))
def Gmap(k):
    h=psi2(k); U=[(a,b),(c,d),(-a-c,-b-d)]
    t=[T3(ue[0]+h[0]/3,ue[1]+h[1]/3) for ue in U]
    m=[(t[0][j]+t[1][j]+t[2][j])/3 for j in (0,1)]
    return [tr(t[0][0]-m[0]),tr(t[0][1]-m[1]),tr(t[1][0]-m[0]),tr(t[1][1]-m[1])]
def compose(Gm,Hm):  # Gm∘Hm truncated
    s=dict(zip(V,Hm)); return [tr(g.subs(s,simultaneous=True)) for g in Gm]
s3=sp.sqrt(3); lam=R(1,2)+sp.I*s3/2; lamb=R(1,2)-sp.I*s3/2
u,v,U_,V_=sp.symbols('u v ub vb')   # ub=conj(u), vb=conj(v)
def chiral(k):
    G=Gmap(k); M=list(V)
    for _ in range(6): M=compose(G,M)
    f=[sp.expand(M[i]-V[i]) for i in range(4)]
    f=[sp.Add(*[t for t in sp.Add.make_args(fi) if sp.Poly(t,*V).total_degree()==3]) for fi in f]
    # real coords from w (eigen): rho = Re w + Im w/sqrt3, eta = 2 Im w/sqrt3 ; w = rho - eta/2 + i sqrt3/2 eta
    w2=(u-v)/(lam-lamb); w1=u-lam*w2; W2=(U_-V_)/(lamb-lam); W1=U_-lamb*W2
    def re_im(w,W): return ((w+W)/2,(w-W)/(2*sp.I))
    (x1,y1),(x2,y2)=re_im(w1,W1),re_im(w2,W2)
    sub={a:x1+y1/s3,b:2*y1/s3,c:x2+y2/s3,d:2*y2/s3}
    fW=[f[0]-f[1]/2+sp.I*s3/2*f[1], f[2]-f[3]/2+sp.I*s3/2*f[3]]
    uf=sp.expand(sp.expand(fW[0]+lam*fW[1]).subs(sub,simultaneous=True))
    P=sp.Poly(uf,u,v,U_,V_)
    p=sp.nsimplify(sp.simplify(P.coeff_monomial(u**2*U_))); q=sp.nsimplify(sp.simplify(P.coeff_monomial(u*v*V_)))
    other=[sp.simplify(cf) for mo,cf in zip(P.monoms(),P.coeffs()) if mo not in [(2,0,1,0),(1,1,0,1)]]
    return sp.radsimp(p),sp.radsimp(q),[o for o in other if o!=0]
kk=sp.Symbol('k')
Dk=lambda k:(k-2)*(k**2-k+1)
qform=lambda k:-R(2,3)*(k-1)**3/Dk(k)-sp.I*2*s3/9*(2*k**2-3*k+2)/(k**2-k+1)
pform=lambda k:-(k**2-3*k+1)/(6*(k**2-k+1))-sp.I*s3/9*(2*k**2-3*k+2)/(k**2-k+1)
pts=[R(1,2),R(3,2),R(1,4),R(3),R(-1),R(5,3),R(7)]
qD=[];pD=[]
for k in pts:
    p,q,oth=chiral(k)
    print(f"k={k}: q={q}  p={p}  other-monomials-nonzero={len(oth)}  q-qform={sp.simplify(q-qform(k))}  p-pform={sp.simplify(p-pform(k))}")
    qD.append((k,sp.expand(q*Dk(k)))); pD.append((k,sp.expand(p*Dk(k))))
# exact interpolation: q*D polynomial deg<=3 from first 4 points, check rest
polyq=sp.expand(sp.interpolate(qD[:4],kk)); polyp=sp.expand(sp.interpolate(pD[:4],kk))
print("q*D interpolant:",sp.factor(polyq)); print("matches qform*D:",sp.simplify(polyq-sp.expand(qform(kk)*Dk(kk)))==0)
print("checks at extra points:",[sp.simplify(polyq.subs(kk,k)-v)==0 for k,v in qD[4:]])
print("p*D interpolant matches:",sp.simplify(polyp-sp.expand(pform(kk)*Dk(kk)))==0,[sp.simplify(polyp.subs(kk,k)-v)==0 for k,v in pD[4:]])
