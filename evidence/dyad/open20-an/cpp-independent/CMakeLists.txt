cmake_minimum_required(VERSION 3.20)
project(dyad_independent_kernel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(MPFR_INCLUDE_DIR mpfr.h REQUIRED)
find_library(MPFR_LIBRARY mpfr REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
add_executable(dyad_cpp_kernel src/main.cpp)
target_include_directories(dyad_cpp_kernel PRIVATE ${MPFR_INCLUDE_DIR})
target_link_libraries(dyad_cpp_kernel PRIVATE ${MPFR_LIBRARY} ${GMP_LIBRARY})
target_compile_options(dyad_cpp_kernel PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
foreach(bits IN ITEMS 128 256 512)
  add_test(NAME kernel_${bits} COMMAND dyad_cpp_kernel --bits ${bits})
endforeach()
